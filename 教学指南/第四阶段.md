```java
@Transactional
public void follow(long followerId, long targetId) {
    // 0.不能关注自己
    if (followerId == targetId) {
        throw new BusinessException(ErrorCode.BAD_REQUEST, "不能关注自己");
    }
    
    // 1.检查目标用户是否存在
    User target = userMapper.findById(targetId);
    if (target == null) {
        throw new BusinessException(ErrorCode.NOT_FOUND, "用户不存在");
    }
    
    // 2.检查是否已关注
    if (exists(followerId, targetId)) {
        throw new BusinessException(ErrorCode.BAD_REQUEST, "已关注");
    }
    
    // 3.写入关注关系
    followingMapper.insert(followerId, targetId);
    
    // 4.增加计数
    counterService.incrementFollowing(followerId, 1);
    counterService.incrementFollowers(targetId, 1);
    
    // 5.写入Outbox事件（异步传播到Kafka）
    outboxMapper.insert(...)
}
```

```java
boolean hasFavorited(long userId, long postId); // 是否收藏
```

**Bitmap 设计**：
```
key: counter:like:{postId}     // Bitmap，bit位置=userId % 10000000
key: counter:fav:{postId}      // Bitmap，收藏用户集合
key: counter:following:{userId} // Bitmap，关注列表
key: counter:follower:{userId}  // Bitmap，粉丝列表
```

**Lua 脚本**（原子操作 + 一致性校验）：
```lua
-- increment.lua
local key = KEYS[1]
local userId = tonumber(ARGV[1])
local delta = tonumber(ARGV[2])

-- 计算bitmap位置
local pos = userId % 10000000

-- 获取当前值
local current = redis.call('GETBIT', key, pos)

-- 更新
local newValue = current + delta
if newValue < 0 then newValue = 0 end

redis.call('SETBIT', key, pos, newValue)
return newValue
```

```java
private static final int HOT_THRESHOLD = 1000; // 1000次/分钟
private static final long HOT_TIME_WINDOW = 60_000L; // 60秒

public boolean isHotKey(long postId) {
    String key = "hot:post:" + postId;
    String count = redis.opsForValue().get(key);
    if (count == null) return false;
    return Long.parseLong(count) >= HOT_THRESHOLD;
}

public void recordAccess(long postId) {
    String key = "hot:post:" + postId;
    Long increment = redis.opsForValue().increment(key);
    if (increment == 1) {
        redis.expire(key, 60);
    }
}
```

### 2.1 关注按钮组件

**路径**：`源代码/zhiguang_fe/src/components/common/FollowButton.tsx`

```typescript
type FollowButtonProps = {
    targetId: number;
    isFollowing: boolean;
    followerCount: number;
    onFollowChange?: (isFollowing: boolean, newCount: number) => void;
};

const FollowButton = ({targetId, isFollowing, followerCount, onFollowChange}: FollowButtonProps) => {
    const [following, setFollowing] = useState(isFollowing);
    const [count, setCount] = useState(followerCount);
    const [loading, setLoading] = useState(false);
    
    const handleClick = async () => {
        setLoading(true);
        try {
            if (following) {
                await relationService.unfollow(targetId);
                setFollowing(false);
                setCount(c => c - 1);
            } else {
                await relationService.follow(targetId);
                setFollowing(true);
                setCount(c => c + 1);
            }
            onFollowChange?.(following, count);
        } finally {
            setLoading(false);
        }
    };
    
    return (
        <button 
            className={following ? 'following' : 'follow'}
            onClick={handleClick}
            disabled={loading}
        >
            {following ? '已关注' : '关注'}
            {count > 0 && <span className="count">{count}</span>}
        </button>
    );
};
```

### 2.4 首页 Feed 页面

**路径**：`源代码/zhiguang_fe/src/pages/HomePage.tsx`

```typescript
const HomePage = () => {
    const {user} = useAuth();
    const [feedType, setFeedType] = useState<'recommend' | 'following'>('recommend');
    const [items, setItems] = useState<FeedItem[]>([]);
    const [page, setPage] = useState(1);
    const [hasMore, setHasMore] = useState(true);
    const [loading, setLoading] = useState(false);
    
    const loadFeed = async (reset: boolean = false) => {
        if (loading) return;
        setLoading(true);
        try {
            const resp = feedType === 'recommend'
                ? await feedService.getFeed(reset ? 1 : page, 20, tokens.accessToken)
                : await feedService.getFollowingFeed(reset ? 1 : page, 20, tokens.accessToken);
            
            if (reset) {
                setItems(resp.items ?? []);
                setPage(resp.page ?? 1);
            } else {
                setItems(prev => [...prev, ...(resp.items ?? [])]);
            }
            setHasMore(!!resp.hasMore);
        } finally {
            setLoading(false);
        }
    };
    
    useEffect(() => {
        void loadFeed(true);
    }, [feedType]);
    
    // 下拉加载更多
    const handleScroll = (e: React.UIEvent<HTMLDivElement>) => {
        const {scrollTop, clientHeight, scrollHeight} = e.currentTarget;
        if (scrollHeight - scrollTop - clientHeight < 200 && hasMore && !loading) {
            setPage(p => p + 1);
            void loadFeed();
        }
    };
    
    return (
        <div className="home-page" onScroll={handleScroll}>
            <Tabs value={feedType} onChange={setFeedType}>
                <Tab value="recommend">推荐</Tab>
                <Tab value="following">关注</Tab>
            </Tabs>
            
            <div className="feed-list">
                {items.map(item => (
                    <CourseCard key={item.id} {...item} />
                ))}
            </div>
            
            {loading && <LoadingSpinner />}
            {!hasMore && <div className="no-more">没有更多了</div>}
        </div>
    );
};
```

## 四、常见问题

### Q1: 点赞数不准确？
**A**: 检查三点：
1. Lua 脚本是否原子执行
2. Bitmap 位移计算是否一致（userId % 10000000）
3. 是否存在并发覆盖

### Q2: 热键检测失效？
**A**: 检查：
1. `hot:post:{id}` key 是否设置 TTL
2. 计数窗口是否为 60 秒
3. 阈值是否为 1000

### Q3: 关注关系同步延迟？
**A**: 这是设计预期。关注写入 Outbox 后通过 Canal→Kafka 异步传播，存在秒级延迟。
